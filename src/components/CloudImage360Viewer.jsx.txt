import React, { useState, useEffect } from 'react';
import React360Viewer from 'react-360-product-viewer';

const CloudImage360Viewer = ({ 
  images = [],
  autoplay = true,
  autoplaySpeed = 5,
  width = '100%',
  height = '100%'
}) => {
  const [isLoaded, setIsLoaded] = useState(false);
  const [error, setError] = useState(false);

  // Preload images
  useEffect(() => {
    if (images && images.length > 0) {
      // Preload the first few images
      const preloadImages = images.slice(0, 3).map((url) => {
        const img = new Image();
        img.src = url;
        return img;
      });
      
      Promise.all(preloadImages.map(img => {
        return new Promise((resolve) => {
          img.onload = resolve;
          img.onerror = resolve;
        });
      })).then(() => {
        setIsLoaded(true);
      });
    }
  }, [images]);

  // If no images, show placeholder
  if (!images || images.length === 0) {
    return (
      <div style={{ 
        width, 
        height, 
        display: 'flex', 
        alignItems: 'center', 
        justifyContent: 'center',
        background: 'radial-gradient(ellipse at center, #1a1a2e 0%, #0f0f1a 100%)',
        color: 'rgba(255,255,255,0.1)'
      }}>
        <span>📷 No Images</span>
      </div>
    );
  }

  // If error, show fallback
  if (error) {
    return (
      <div style={{ 
        width, 
        height, 
        display: 'flex', 
        alignItems: 'center', 
        justifyContent: 'center',
        background: 'radial-gradient(ellipse at center, #1a1a2e 0%, #0f0f1a 100%)'
      }}>
        <div style={{ textAlign: 'center', color: 'rgba(255,255,255,0.3)' }}>
          <div style={{ fontSize: '2rem', marginBottom: '10px' }}>🔄</div>
          <div style={{ fontSize: '0.8rem' }}>360° View</div>
        </div>
      </div>
    );
  }

  // Determine base URL and count from images
  const firstImage = images[0];
  let baseUrl = '';
  let count = images.length;
  let fileType = 'jpg';

  // Try to extract base URL from first image
  if (firstImage) {
    // Remove filename from URL
    const parts = firstImage.split('/');
    const filename = parts.pop();
    baseUrl = parts.join('/') + '/';
    
    // Get file extension
    const extParts = filename.split('.');
    if (extParts.length > 1) {
      fileType = extParts.pop();
    }
  }

  // Check if images are sequentially named (angle-1, angle-2, etc.)
  const hasSequentialNaming = images.every((url, index) => {
    const expectedUrl = `${baseUrl}angle-${index + 1}.${fileType}`;
    return url.includes(`angle-${index + 1}`);
  });

  return (
    <div style={{ 
      width, 
      height, 
      position: 'relative',
      background: 'radial-gradient(ellipse at center, #1a1a2e 0%, #0f0f1a 100%)'
    }}>
      {!isLoaded ? (
        <div style={{
          position: 'absolute',
          top: '50%',
          left: '50%',
          transform: 'translate(-50%, -50%)',
          zIndex: 5
        }}>
          <div style={{
            fontSize: '2rem',
            animation: 'spin 1.2s ease-in-out infinite'
          }}>🔄</div>
        </div>
      ) : (
        <React360Viewer
          imagesBaseUrl={baseUrl}
          imagesCount={count}
          imagesFiletype={fileType}
          imageFilenamePrefix="angle-"
          autoplay={autoplay}
          autoplaySpeed={autoplaySpeed}
          mouseDragSpeed={20}
          width="100%"
          height="100%"
        />
      )}
      
      {/* Style for spin animation */}
      <style>{`
        @keyframes spin {
          0% { transform: rotate(0deg) scale(1); }
          50% { transform: rotate(180deg) scale(1.1); }
          100% { transform: rotate(360deg) scale(1); }
        }
      `}</style>
    </div>
  );
};

export default CloudImage360Viewer;
